@dataclass(frozen=True)
class PreparedTransaction:
    """Signed transaction which has not been broadcast yet.

    :param inputs: Transaction inputs (including private keys used to sign them)
    :param outputs: Destination output and optional change output sent back to the source address
    :param change_added_to_fee: Change below the dust threshold, which was added to the fee
    """

    raw_transaction: str
    inputs: List[Dict[str, Any]]
    outputs: List[Dict[str, Any]]
    input_satoshis: int
    fee: int
    change_added_to_fee: int = 0


def build_transaction(
    coin: Coin,
    inputs: List[Dict[str, Any]],
    destination_address: str,
    source_address: str,
    input_satoshis: int,
    satoshis: int,
    fee: int,
) -> PreparedTransaction:
    """Create signed transaction, the change is sent back to the source address."""
    outputs = [{'address': destination_address, 'satoshis': satoshis}]
    change = input_satoshis - satoshis - fee
    change_added_to_fee = 0
    dust_threshold = get_dust_threshold(address_to_script(source_address, coin))
    if change >= dust_threshold:
        outputs.append({'address': source_address, 'satoshis': change})
    elif change > 0:
        change_added_to_fee = change
        logger.info(
            'Change %s is below dust threshold %s and was added to the fee', change, dust_threshold
        )
    history_block = {}
    if 'check_block_at_height' in coin.get('params', {}):
        history_block = get_api(coin).get_history_block()
    transaction = Transaction(coin, inputs, outputs, check_block_at_height=history_block)
    return PreparedTransaction(
        transaction.create(), inputs, outputs, input_satoshis, fee, change_added_to_fee
    )


def broadcast_prepared(
    coin: Coin, prepared: PreparedTransaction, broadcast: Optional[Broadcast] = None
) -> str:
    """Broadcast the prepared transaction.

    :param broadcast: Function sending raw transaction (hex) and returning the result, by default
                      the transaction is sent to the coin block explorer API
    """
    if broadcast is None:
        return get_api(coin).send_raw_transaction(prepared.raw_transaction)
    return broadcast(prepared.raw_transaction)


def prepare_from_private_keys(
    coin: Coin,
    wif_private_keys: Optional[List[str]],
    input_addresses: Optional[List[str]],
    unlocking_scripts: Optional[List[str]],
    destination_address: str,
    satoshis: int,
    fee: int,
    minimum_input_threshold: Optional[int] = None,
    maximum_input_threshold: Optional[int] = None,
    limit_inputs: Optional[int] = None,
    address_type: str = P2PKH,
) -> PreparedTransaction:
    if not input_addresses and not wif_private_keys:
        raise ValueError('Missing required parameters input_addresses or wif_private_keys')
    utxo_sources = []
    if input_addresses:
        utxo_sources.append(
            get_utxo_from_addresses(
                coin,
                input_addresses,
                unlocking_scripts,
                minimum_input_threshold,
                maximum_input_threshold,
                limit_inputs,
            )
        )
    if wif_private_keys:
        utxo_sources.append(
            get_utxo_from_private_keys(
                coin,
                wif_private_keys,
                address_type,
                minimum_input_threshold,
                maximum_input_threshold,
                limit_inputs,
            )
        )
    utxos = itertools.chain.from_iterable(utxo_sources)
    return prepare_utxos(coin, utxos, destination_address, satoshis, fee)


def send_from_private_keys(
    coin: Coin,
    wif_private_keys: Optional[List[str]],
    input_addresses: Optional[List[str]],
    unlocking_scripts: Optional[List[str]],
    destination_address: str,
    satoshis: int,
    fee: int,
    minimum_input_threshold: Optional[int] = None,
    maximum_input_threshold: Optional[int] = None,
    limit_inputs: Optional[int] = None,
    broadcast: Optional[Broadcast] = None,
    address_type: str = P2PKH,
) -> str:
    prepared = prepare_from_private_keys(
        coin,
        wif_private_keys,
        input_addresses,
        unlocking_scripts,
        destination_address,
        satoshis,
        fee,
        minimum_input_threshold,
        maximum_input_threshold,
        limit_inputs,
        address_type,
    )
    return broadcast_prepared(coin, prepared, broadcast)


def prepare_utxos(
    coin: Coin,
    utxos: Iterable[Utxo],
    destination_address: str,
    satoshis: int,
    fee: int,
) -> PreparedTransaction:
    inputs = []
    input_satoshis = 0
    for utxo in utxos:
        if not utxo:
            continue
        input_satoshis += utxo['satoshis']
        inputs.append(
            {
                'transaction_id': reverse_byte_hex(utxo['txid']),
                'output_index': utxo['vout'],
                'locking_script': utxo['scriptPubKey'],
                'satoshis': utxo['satoshis'],
                'private_key': utxo.get('private_key'),
                'unlocking_script': utxo.get('unlocking_script', ''),
            }
        )
        logger.debug(
            '%s. address: %s, input: %s, satoshis: %s',
            len(inputs),
            utxo['source_address'],
            utxo['txid'],
            utxo['satoshis'],
        )
        if input_satoshis >= satoshis + fee:
            return build_transaction(
                coin,
                inputs,
                destination_address,
                utxo['source_address'],
                input_satoshis,
                satoshis,
                fee,
            )
    raise InsufficientFundsError(
        f'Not enough funds in address(es).\n'
        f'Sum of inputs is {input_satoshis} which is less than '
        f'{satoshis + fee} ({satoshis} + {fee})'
    )


def send_utxos(
    coin: Coin,
    utxos: Iterable[Utxo],
    destination_address: str,
    satoshis: int,
    fee: int,
    broadcast: Optional[Broadcast] = None,
) -> str:
    prepared = prepare_utxos(coin, utxos, destination_address, satoshis, fee)
    return broadcast_prepared(coin, prepared, broadcast)
```

```python
"""Wallet keeping private keys encrypted in a single file."""

import json
import uuid
from dataclasses import asdict, dataclass, field
from datetime import datetime, timezone
from functools import cached_property
from typing import Any, Dict, Iterable, List, Optional

from multicrypto.address import (
    convert_private_key_to_coin_address,
    convert_private_key_to_wif_format,
    get_private_key_from_wif_format,
    validate_address_type,
)
from multicrypto.base58 import BASE58_ALPHABET, base58_to_bytes
from multicrypto.coins import coins
from multicrypto.ellipticcurve import secp256k1
from multicrypto.utils import double_sha256


def get_coin(coin_symbol: str) -> Dict[str, Any]:
    if coin_symbol not in coins:
        raise ValueError(f'Unknown coin {coin_symbol}')
    return coins[coin_symbol]


def check_wif_private_key(wif_private_key: str, coin_symbol: str) -> None:
    """Check characters, checksum, coin prefix and range of the private key in WIF format."""
    if not wif_private_key or set(wif_private_key) - set(BASE58_ALPHABET):
        raise ValueError('Private key in WIF format can contain only base58 characters')
    data = base58_to_bytes(wif_private_key)
    payload, checksum = data[:-4], data[-4:]
    if len(data) < 5 or double_sha256(payload).digest()[:4] != checksum:
        raise ValueError('Incorrect checksum of the private key')
    prefix = get_coin(coin_symbol)['secret_prefix_bytes']
    if not payload.startswith(prefix):
        raise ValueError(f'Private key is not in WIF format of {coin_symbol}')
    key_bytes = payload[len(prefix) :]
    if len(key_bytes) == 33 and key_bytes[-1] == 1:  # flag of the compressed public key
        key_bytes = key_bytes[:-1]
    if len(key_bytes) != 32 or not 0 < int.from_bytes(key_bytes, 'big') < secp256k1.n:
        raise ValueError('Incorrect private key')


def check_entry(coin_symbol: str, wif_private_key: str, address_type: str) -> str:
    """Validate wallet entry data and return its address."""
    coin = get_coin(coin_symbol)
    validate_address_type(address_type, coin)
    check_wif_private_key(wif_private_key, coin_symbol)
    private_key, compressed = get_private_key_from_wif_format(wif_private_key)
    return convert_private_key_to_coin_address(private_key, coin, compressed, address_type)


def generate_wif(coin_symbol: str, private_key: Optional[int] = None) -> str:
    """Return new random private key (or the given one) in WIF format with compressed public key."""
    if private_key is None:
        private_key = secp256k1.gen_private_key()
    return convert_private_key_to_wif_format(
        private_key, get_coin(coin_symbol)['secret_prefix_bytes']
    )


def _now() -> str:
    return datetime.now(timezone.utc).strftime('%Y-%m-%dT%H:%M:%SZ')


@dataclass
class WalletEntry:
    """Private key used with the address of the given type of the coin."""

    label: str
    coin: str
    wif: str = field(repr=False)
    address_type: str
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    created: str = field(default_factory=_now)

    @cached_property
    def address(self) -> str:
        private_key, compressed = get_private_key_from_wif_format(self.wif)
        return convert_private_key_to_coin_address(
            private_key, coins[self.coin], compressed, self.address_type
        )


class Wallet:
    """Decrypted content of the wallet file."""

    def __init__(self, entries: Iterable[WalletEntry] = ()):
        self.entries: List[WalletEntry] = list(entries)

    def get(self, entry_id: str) -> WalletEntry:
        for entry in self.entries:
            if entry.id == entry_id:
                return entry
        raise KeyError(entry_id)

    def add_wif(

                coin_symbol,
                wif_private_key,
                address_type,
            ):
                raise ValueError(
                    f'Address {entry.address} is already in the wallet as "{entry.label}"'
                )
        entry = WalletEntry(label, coin_symbol, wif_private_key, address_type)
        self.entries.append(entry)
        return entry

    def remove(self, entry_id: str) -> None:
        self.entries.remove(self.get(entry_id))

    def to_json(self) -> bytes:
        return json.dumps({'entries': [asdict(entry) for entry in self.entries]}).encode()

    @classmethod
    def from_json(cls, data: bytes) -> 'Wallet':
        return cls(WalletEntry(**entry) for entry in json.loads(data)['entries'])

def check_new_password(password: str) -> None:
    if len(password) < MIN_PASSWORD_LENGTH:
        raise ValueError(f'Password must have at least {MIN_PASSWORD_LENGTH} characters')


def derive_key(password: str, salt: bytes, opslimit: int, memlimit: int) -> bytes:
    """Derive the encryption key from the password, which is normalized, so the same password
    entered in different ways (e.g. composed or decomposed accented letters) gives the same key."""
    password_bytes = unicodedata.normalize('NFKC', password).encode()
    return pwhash.argon2id.kdf(KEY_SIZE, password_bytes, salt, opslimit=opslimit, memlimit=memlimit)


class WalletFile:
    """Encrypted wallet file, the key derived from the password is kept until it is locked."""

    def __init__(
        self,
        path: str = DEFAULT_PATH,
        opslimit: int = DEFAULT_OPSLIMIT,
        memlimit: int = DEFAULT_MEMLIMIT,
    ):
        self.path = path
        self.opslimit = opslimit
        self.memlimit = memlimit
        self._salt: Optional[bytes] = None
        self._key: Optional[bytes] = None

    @property
    def is_unlocked(self) -> bool:
        return self._key is not None

    def exists(self) -> bool:
        return os.path.exists(self.path)

    def create(self, password: str) -> Wallet:
        """Create new empty wallet file encrypted with the key derived from the password."""
        check_new_password(password)
        if self.exists():
            raise WalletError(f'Wallet file {self.path} already exists')
        os.makedirs(os.path.dirname(os.path.abspath(self.path)), mode=0o700, exist_ok=True)
        self._salt = os.urandom(pwhash.argon2id.SALTBYTES)
        self._key = derive_key(password, self._salt, self.opslimit, self.memlimit)
        wallet = Wallet()
        self.save(wallet)
        return wallet

    def unlock(self, password: str) -> Wallet:
        """Decrypt the wallet file, raise WalletError when the password is wrong."""
        with open(self.path, 'rb') as file:
            data = file.read()
        if len(data) < HEADER_SIZE + TAG_SIZE:
            raise WalletError(UNSUPPORTED_FORMAT)
        header = data[:HEADER_SIZE]
        magic, version, opslimit, memlimit, salt, nonce = struct.unpack(HEADER_FORMAT, header)
        if (
            magic != MAGIC
            or version != VERSION
            or not pwhash.argon2id.OPSLIMIT_MIN <= opslimit <= pwhash.argon2id.OPSLIMIT_MAX
            or not pwhash.argon2id.MEMLIMIT_MIN <= memlimit <= pwhash.argon2id.MEMLIMIT_SENSITIVE
        ):
            raise WalletError(UNSUPPORTED_FORMAT)
        key = derive_key(password, salt, opslimit, memlimit)
        try:
            content = bindings.crypto_aead_xchacha20poly1305_ietf_decrypt(
                data[HEADER_SIZE:], header, nonce, key
            )
        except CryptoError:
            raise WalletError(WRONG_PASSWORD) from None
        wallet = Wallet.from_json(content)
        self.opslimit, self.memlimit, self._salt, self._key = opslimit, memlimit, salt, key
        return wallet

    def verify_password(self, password: str) -> bool:
        """Check the password of the unlocked wallet, e.g. before showing a private key."""
        if self._key is None:
            raise WalletError('Wallet is locked')
        key = derive_key(password, self._salt, self.opslimit, self.memlimit)
        return compare_digest(key, self._key)

    def save(self, wallet: Wallet) -> None:
        """Encrypt the wallet using new nonce and atomically replace the wallet file."""
        if self._key is None:
            raise WalletError('Wallet is locked')
        nonce = os.urandom(NONCE_SIZE)
        header = struct.pack(
            HEADER_FORMAT, MAGIC, VERSION, self.opslimit, self.memlimit, self._salt, nonce
        )
        ciphertext = bindings.crypto_aead_xchacha20poly1305_ietf_encrypt(
            wallet.to_json(), header, nonce, self._key
        )
        temporary_path = f'{self.path}.tmp'
        if os.path.exists(temporary_path):
            os.remove(temporary_path)
        flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL | getattr(os, 'O_BINARY', 0)
        descriptor = os.open(temporary_path, flags, 0o600)
        try:
            with os.fdopen(descriptor, 'wb') as file:
                file.write(header + ciphertext)
                file.flush()
                os.fsync(file.fileno())
            os.replace(temporary_path, self.path)
        except BaseException:
            if os.path.exists(temporary_path):
                os.remove(temporary_path)
            raise

    def change_password(self, old_password: str, new_password: str) -> Wallet:
        """Encrypt the wallet with new salt and the key derived from the new password."""
        check_new_password(new_password)
        wallet = self.unlock(old_password)
        old_salt, old_key = self._salt, self._key
        self._salt = os.urandom(pwhash.argon2id.SALTBYTES)
        self._key = derive_key(new_password, self._salt, self.opslimit, self.memlimit)
        try:
            self.save(wallet)
        except BaseException:
            self._salt, self._key = old_salt, old_key
            raise
        return wallet

    def lock(self) -> None:
        self._salt = None
        self._key = None

    address = text.strip()
    if not address:
        raise ValueError('Enter the address')
    try:
        validate_address(address, coin_symbol)
    except ValueError:
        raise
    except Exception:  # base58 decoding raises KeyError, unknown prefix raises Exception
        raise ValueError(f'Incorrect address {address} for {coin_symbol}') from None
    return address


def check_password_confirmation(password: str, confirmation: str) -> None:
    check_new_password(password)
    if password != confirmation:
        raise ValueError('Passwords do not match')


def check_sources(entries: List[WalletEntry]) -> Tuple[str, str]:
    """Check that the entries can be spent in one transaction, return coin and address type."""
    if not entries:
        raise ValueError('Select at least one source address')
    kinds = {(entry.coin, entry.address_type) for entry in entries}
    if len(kinds) > 1:
        raise ValueError('All source addresses must have the same coin and address type')
    return kinds.pop()


def parse_nodes(text: str, default_port: int) -> List[PeerAddress]:
    """Parse nodes in host[:port] format separated with commas or whitespace."""
    return [parse_node_address(node, default_port) for node in text.replace(',', ' ').split()]


def extract_txid(result: str) -> str:
    """Return the transaction id from the broadcast result, which is JSON on success."""
    try:
        return json.loads(result)['txid']
    except (ValueError, KeyError, TypeError):
        raise ValueError(f'Transaction was not accepted: {result}') from None


def coins_with_api() -> List[str]:
    return sorted(symbol for symbol, coin in coins.items() if coin.get('apis'))


def entry_description(entry: WalletEntry) -> str:
    return f'{entry.label} · {entry.coin} {entry.address_type} · {entry.address}'


def describe_transaction(
    prepared: PreparedTransaction,
    coin_symbol: str,
    source_addresses: List[str],
    broadcast_target: str,
) -> List[Tuple[str, str]]:
    """Rows of the summary shown before the transaction is broadcast."""
    destination, *change = prepared.outputs
    fee = f'{format_amount(prepared.fee)} {coin_symbol}'
    if prepared.change_added_to_fee:
        fee += f' + {prepared.change_added_to_fee} satoshis of change below the dust threshold'
    if change:
        change_text = (
            f'{format_amount(change[0]["satoshis"])} {coin_symbol} to {change[0]["address"]}'
        )
    else:
        change_text = 'none'
    return [
        ('From', '\n'.join(source_addresses)),
        ('To', destination['address']),
        ('Amount', f'{format_amount(destination["satoshis"])} {coin_symbol}'),
        ('Fee', fee),
        ('Change', change_text),
        (
            'Inputs',
            f'{len(prepared.inputs)} ({format_amount(prepared.input_satoshis)} {coin_symbol})',
        ),
        ('Size', f'{len(prepared.raw_transaction) // 2} bytes'),
        ('Broadcast', broadcast_target),
    ]


def describe_error(error: BaseException) -> str:
    """Return the message shown to the user, unexpected errors are logged with the traceback."""
    if isinstance(error, EXPECTED_ERRORS):
        return str(error) or type(error).__name__
    logger.error('Unexpected error', exc_info=error)
    return f'Unexpected error {type(error).__name__}: {error}'

```python
"""Running slow operations (network requests, key derivation) outside of the tkinter thread."""

import logging
import queue
import threading
from typing import Any, Callable

logger = logging.getLogger(__name__)

POLL_INTERVAL = 100  # in milliseconds


class Worker:
    """Run functions in background threads and call the callbacks in the tkinter thread, since
    tkinter widgets must not be used from other threads."""

    def __init__(self, root):
        self.root = root
        self.pending = 0
        self._results = queue.Queue()
        self.root.after(POLL_INTERVAL, self._poll)

    def run(
        self,
        func: Callable[[], Any],
        on_success: Callable[[Any], None],
        on_error: Callable[[Exception], None],
    ) -> None:
        def target():
            try:
                result = func()
            except Exception as error:  # passed to on_error in the tkinter thread
                self._results.put((on_error, error))
            else:
                self._results.put((on_success, result))

        self.pending += 1
        threading.Thread(target=target, daemon=True).start()

    def _poll(self) -> None:
        try:
            while True:
                try:
                    callback, value = self._results.get_nowait()
                except queue.Empty:
                    break
                self.pending -= 1
                try:
                    callback(value)
                except Exception:
                    logger.exception('Error in the callback of the background task')
        finally:
            self.root.after(POLL_INTERVAL, self._poll)

        self.nodes = self._add_row(advanced, 4, 'Nodes host[:port] (optional)', width=74)
        self.button = ttk.Button(self.frame, text='Prepare transaction…', command=self.prepare)
        self.button.pack(anchor='e')

    @staticmethod
    def _add_row(parent, row: int, text: str, width: int) -> ttk.Entry:
        ttk.Label(parent, text=text).grid(row=row, column=0, sticky='w', padx=(0, 8), pady=2)
        entry = ttk.Entry(parent, width=width)
        entry.grid(row=row, column=1, sticky='w', pady=2)

        show_entries(self.sources, self.app.wallet.entries)

    def clear(self):
        self.sources.delete(*self.sources.get_children())
        self.destination.delete(0, 'end')
        self.amount.delete(0, 'end')

    def select_source(self, entry):
        self.sources.selection_set([entry.id])
        self.app.notebook.select(self.frame)

    def prepare(self):
        try:
            entries = [self.app.wallet.get(item) for item in self.sources.selection()]
            coin_symbol, address_type = check_sources(entries)
            coin = coins[coin_symbol]
            if not coin.get('apis'):
                raise ValueError(f'No API has been defined for the coin {coin_symbol}')
            destination = parse_address(self.destination.get(), coin_symbol)
            satoshis = parse_amount(self.amount.get())
            fee = parse_int(self.fee.get(), 'Fee', minimum=0)
            minimum, maximum, limit = parse_thresholds(
                self.minimum.get(), self.maximum.get(), self.limit.get()
            )
            broadcast, target = self._get_broadcast(coin, coin_symbol)
        except ValueError as error:
            self.app.show_error(error)
            return
        wif_private_keys = [entry.wif for entry in entries]
        source_addresses = [entry.address for entry in entries]

        def prepare_transaction():
            return prepare_from_private_keys(
                coin=coin,
                wif_private_keys=wif_private_keys,
                input_addresses=None,
                unlocking_scripts=None,
                destination_address=destination,
                satoshis=satoshis,
                fee=fee,

                maximum_input_threshold=maximum,
                limit_inputs=limit,
                address_type=address_type,
            )

        self.app.worker.run(
            prepare_transaction,
            lambda prepared: self._confirm(
                coin, coin_symbol, prepared, source_addresses, broadcast, target
            ),
            self._failed,
        )

    def _get_broadcast(self, coin, coin_symbol: str):
        """Return function broadcasting the transaction (None for the API) and its description."""
        nodes_text = self.nodes.get()
        if not self.p2p.get() and not nodes_text.strip():
            return None, 'block explorer API'
        if 'network' not in coin:
            raise ValueError(f'No peer-to-peer network has been defined for the coin {coin_symbol}')
        nodes = parse_nodes(nodes_text, coin['network']['port'])

        def broadcast(raw_transaction):
            return json.dumps({'txid': broadcast_transaction(coin, raw_transaction, nodes)})

        return broadcast, 'peer-to-peer network'

    def _confirm(self, coin, coin_symbol, prepared, source_addresses, broadcast, target):
        self.button.state(['!disabled'])
        rows = describe_transaction(prepared, coin_symbol, source_addresses, target)
        if not confirm(self.frame, 'Confirm transaction', rows, 'Send'):
            return
        self.button.state(['disabled'])
        self.app.worker.run(
            lambda: extract_txid(broadcast_prepared(coin, prepared, broadcast)),
            self._sent,
            self._failed,
        )

    def _sent(self, txid: str):
        self.button.state(['!disabled'])
        self.amount.delete(0, 'end')
        TextDialog(self.frame, 'Transaction sent', 'Transaction id', txid, copy=True)

    def _failed(self, error):
        self.button.state(['!disabled'])
        self.app.show_error(error)

"""Main window: wallet file handling, tabs and automatic locking."""

import os
import time
import tkinter as tk
from tkinter import filedialog, messagebox, ttk
from typing import Optional

from multicrypto.gui.balance_tab import BalanceTab
from multicrypto.gui.dialogs import ask_password
from multicrypto.gui.forms import describe_error
from multicrypto.gui.send_tab import SendTab
from multicrypto.gui.wallet_tab import WalletTab
from multicrypto.gui.worker import Worker
from multicrypto.wallet import MIN_PASSWORD_LENGTH, Wallet, WalletFile

AUTO_LOCK_SECONDS = 5 * 60
WALLET_FILE_TYPES = [('multicrypto wallet', '*.mcw'), ('All files', '*')]


class MainWindow:
    def __init__(
        self, root: tk.Tk, wallet_file: WalletFile, auto_lock_seconds: int = AUTO_LOCK_SECONDS
    ):
        self.root = root
        self.wallet_file = wallet_file
        self.auto_lock_seconds = auto_lock_seconds
        self.wallet: Optional[Wallet] = None
        self.last_activity = time.monotonic()
        self.worker = Worker(root)
        root.title('multicrypto')
        root.minsize(800, 520)
        self._create_menu()
        self.status = tk.StringVar()
        ttk.Label(root, textvariable=self.status, anchor='w').pack(
            side='bottom', fill='x', padx=8, pady=(0, 6)
        )
        self.notebook = ttk.Notebook(root)
        self.wallet_tab = WalletTab(self.notebook, self)
        self.balance_tab = BalanceTab(self.notebook, self)
        self.send_tab = SendTab(self.notebook, self)
        self.tabs = (self.wallet_tab, self.balance_tab, self.send_tab)
        for tab, text in zip(self.tabs, ('Wallet', 'Balance', 'Send'), strict=True):
            self.notebook.add(tab.frame, text=text)
        self.locked_screen = ttk.Frame(root, padding=24)
        self.locked_message = tk.StringVar()
        ttk.Label(self.locked_screen, textvariable=self.locked_message).pack(pady=(40, 12))
        self.unlock_button = ttk.Button(self.locked_screen, text='Unlock…', command=self.unlock)
        self.unlock_button.pack(pady=4)
        ttk.Button(self.locked_screen, text='Create new wallet…', command=self.new_wallet).pack(
            pady=4
        )
        ttk.Button(self.locked_screen, text='Open wallet file…', command=self.open_wallet).pack(
            pady=4
        )
        for sequence in ('<KeyPress>', '<ButtonPress>', '<Motion>'):
            root.bind_all(sequence, self._touch, add='+')
        self._show_locked()
        self.tick()

    def _create_menu(self):
        menubar = tk.Menu(self.root)
        file_menu = tk.Menu(menubar, tearoff=False)
        file_menu.add_command(label='New wallet…', command=self.new_wallet)
        file_menu.add_command(label='Open wallet…', command=self.open_wallet)
        file_menu.add_command(label='Change password…', command=self.change_password)
        file_menu.add_command(label='Lock', command=self.lock)
        file_menu.add_separator()
        file_menu.add_command(label='Quit', command=self.root.destroy)
        menubar.add_cascade(label='File', menu=file_menu)
        self.root.config(menu=menubar)

    def start(self):
        """Ask for the password when the wallet file exists."""
        if self.wallet_file.exists():
            self.unlock()

    def _touch(self, _event=None):
        self.last_activity = time.monotonic()

    def tick(self):
        """Update the status and lock the wallet after inactivity, called every second."""
        if self.wallet is not None:
            remaining = self.auto_lock_seconds - (time.monotonic() - self.last_activity)
            if remaining <= 0:
                self.lock()
            else:
                minutes, seconds = divmod(int(remaining), 60)
                self.status.set(
                    f'{self.wallet_file.path} · unlocked · locks in {minutes}:{seconds:02d}'
                )
        self.root.after(1000, self.tick)

    def _show_locked(self):
        self.notebook.pack_forget()
        if self.wallet_file.exists():
            self.locked_message.set(f'Wallet {self.wallet_file.path} is locked')
            self.unlock_button.state(['!disabled'])
        else:
            self.locked_message.set(
                f'Wallet file {self.wallet_file.path} does not exist.\n'
                'Create a new wallet or open another wallet file.'
            )
            self.unlock_button.state(['disabled'])
        self.status.set(f'{self.wallet_file.path} · locked')
        self.locked_screen.pack(fill='both', expand=True)

    def show_wallet(self, wallet: Wallet):
        self.root.config(cursor='')
        self.wallet = wallet
        self._touch()
        self.locked_screen.pack_forget()
        self.notebook.pack(fill='both', expand=True, padx=8, pady=8)
        for tab in self.tabs:
            tab.refresh()

    def lock(self):
        self.wallet_file.lock()
        self.wallet = None
        for tab in self.tabs:
            tab.clear()
        self._show_locked()

    def _derive_key(self, message: str, func, on_success):
        """Run operation deriving the key from the password (about a second) in background."""
        self.status.set(message)
        self.root.config(cursor='watch')

        def failed(error):
            self.root.config(cursor='')
            if self.wallet is None:
                self._show_locked()
            self.show_error(error)

        self.worker.run(func, on_success, failed)

    def unlock(self):
        password = ask_password(
            self.root, 'Unlock wallet', f'Password of the wallet {self.wallet_file.path}'
        )
        if password is not None:
            wallet_file = self.wallet_file
            self._derive_key('Unlocking…', lambda: wallet_file.unlock(password), self.show_wallet)

    def new_wallet(self):
        if self.wallet_file.exists():
            path = filedialog.asksaveasfilename(
                parent=self.root,
                title='New wallet file',
                initialdir=os.path.dirname(os.path.abspath(self.wallet_file.path)),
                initialfile='wallet.mcw',
                defaultextension='.mcw',
                filetypes=WALLET_FILE_TYPES,
            )
            if not path:
                return
        else:
            path = self.wallet_file.path
        password = ask_password(
            self.root,
            'New wallet',
            f'Password of the new wallet {path}\n(at least {MIN_PASSWORD_LENGTH} characters)',
            new_password=True,
        )
        if password is None:
            return
        self.lock()
        wallet_file = self.wallet_file = WalletFile(path)
        self._derive_key('Creating wallet…', lambda: wallet_file.create(password), self.show_wallet)

    def open_wallet(self):
        path = filedialog.askopenfilename(
            parent=self.root,
            title='Open wallet file',
            initialdir=os.path.dirname(os.path.abspath(self.wallet_file.path)),
            filetypes=WALLET_FILE_TYPES,
        )
        if not path:
            return
        self.lock()
        self.wallet_file = WalletFile(path)
        self._show_locked()
        self.unlock()

    def change_password(self):
        if self.wallet is None:
            return
        old_password = ask_password(self.root, 'Change password', 'Current password')
        if old_password is None:
            return
        new_password = ask_password(
            self.root,
            'Change password',
            f'New password (at least {MIN_PASSWORD_LENGTH} characters)',
            new_password=True,
        )
        if new_password is None:
            return
        wallet_file = self.wallet_file
        self._derive_key(
            'Changing password…',
            lambda: wallet_file.change_password(old_password, new_password),
            self._password_changed,
        )

    def _password_changed(self, wallet: Wallet):
        self.show_wallet(wallet)
        messagebox.showinfo('Change password', 'The password has been changed', parent=self.root)

    def modify_wallet(self, change):
        """Apply the change to the wallet and save it, the change is reverted when saving fails."""
        entries = list(self.wallet.entries)
        result = change(self.wallet)
        try:
            self.wallet_file.save(self.wallet)
        except BaseException:
            self.wallet.entries = entries
            raise
        for tab in self.tabs:
            tab.refresh()
        return result

    def show_error(self, error: BaseException):
        messagebox.showerror('Error', describe_error(error), parent=self.root)

"""Entry point of the multicrypto-gui command."""

import argparse
import logging
import sys

from multicrypto.validators import check_positive


def get_args():
    parser = argparse.ArgumentParser(
        description='Graphical user interface of multicrypto with private keys kept in '
        'encrypted wallet file'
    )
    parser.add_argument(
        '-w',
        '--wallet',
        type=str,
        required=False,
        default=None,
        help='Path of the wallet file, default is ~/.multicrypto/wallet.mcw',
    )
    parser.add_argument(
        '-l',
        '--auto_lock',
        type=check_positive,
        required=False,
        default=5,
        help='Lock the wallet after given number of minutes of inactivity. Default 5 minutes',
    )
    return parser.parse_args()


def main():
    args = get_args()
    logging.basicConfig(level=logging.INFO, format='%(levelname)s %(name)s: %(message)s')
    try:
        import tkinter as tk

        from multicrypto.gui.window import MainWindow
        from multicrypto.wallet import DEFAULT_PATH, WalletFile
    except ModuleNotFoundError as error:
        if error.name == 'nacl':
            sys.exit(
                'multicrypto-gui requires PyNaCl, install it with: pip install "multicrypto[gui]"'
            )
        if error.name in ('tkinter', '_tkinter'):
            sys.exit(
                'multicrypto-gui requires tkinter, install it with the system package manager '
                '(e.g. python3-tk)'
            )
        raise
    root = tk.Tk()
    window = MainWindow(root, WalletFile(args.wallet or DEFAULT_PATH), args.auto_lock * 60)
    root.after_idle(window.start)
    root.mainloop()


if __name__ == '__main__':
    main()

```markdown
### GUI
```bash
multicrypto-gui --wallet=<WALLET FILE> --auto_lock=<MINUTES>
```
The GUI keeps private keys in a single wallet file (by default `~/.multicrypto/wallet.mcw`) encrypted
with XChaCha20-Poly1305 using key derived from the password with Argon2id. Labels and coins of the
addresses are encrypted as well. Tabs:
- **Wallet** - import private key in WIF format, generate new one, copy address, show private key
  (after entering the password again) and remove it,
- **Balance** - unspent outputs of the wallet address or any other address (like `checkaddress`),
- **Send** - send funds from one or more wallet addresses (like `sendcrypto`), the transaction is
  shown for confirmation before it is broadcast.

The wallet is locked after 5 minutes of inactivity (`--auto_lock`). Python cannot reliably remove
data from memory, so private keys of the unlocked wallet stay in memory of the process until it is
locked. Remember to keep a backup of the wallet file and the password - there is no way to recover
the private keys without them.
```
