```python
"""Searching for vanity addresses, which start with the given pattern, in many processes."""

import os

from multicrypto.address import get_address_range, validate_address_type, validate_pattern
from multicrypto.bech32 import CHARSET
from multicrypto.coins import P2PKH, P2SH_P2WPKH, P2WPKH, coins

VANITY_ADDRESS_TYPES = (P2PKH, P2SH_P2WPKH, P2WPKH)


def _bech32_start(coin) -> str:
    return f'{coin["bech32_hrp"]}1q'


def _base58_range(coin_symbol: str, address_type: str):
    key = 'script_prefix_bytes' if address_type == P2SH_P2WPKH else 'address_prefix_bytes'
    return get_address_range(coins[coin_symbol][key])


def validate_vanity_pattern(pattern: str, coin_symbol: str, address_type: str) -> str:
    """Check that an address of the type can start with the pattern, return the pattern
    (lowercase for bech32 addresses)."""
    if address_type not in VANITY_ADDRESS_TYPES:
        raise ValueError(
            f'Vanity addresses of type {address_type} are not supported, '
            f'choose one of {", ".join(VANITY_ADDRESS_TYPES)}'
        )
    coin = coins[coin_symbol]
    validate_address_type(address_type, coin)
    if address_type != P2WPKH:
        validate_pattern(pattern, coin_symbol, address_type == P2SH_P2WPKH)
        return pattern
    pattern = pattern.lower()
    start = _bech32_start(coin)
    if not pattern.startswith(start):
        raise ValueError(f'Pattern must start with {start}')
    for character in pattern[len(start) :]:
        if character not in CHARSET:
            raise ValueError(
                f'Character {character} is not used in bech32 addresses '
                f'(the characters are {"".join(sorted(CHARSET))})'
            )
    return pattern


def expected_attempts(pattern: str, coin_symbol: str, address_type: str) -> int:
    """Average number of addresses checked before one starts with the pattern."""
    if address_type == P2WPKH:
        return 32 ** max(0, len(pattern) - len(_bech32_start(coins[coin_symbol])))
    fixed = len(os.path.commonprefix(_base58_range(coin_symbol, address_type)))
    return 58 ** max(0, len(pattern) - fixed)


def pattern_hint(coin_symbol: str, address_type: str) -> str:
    if address_type == P2WPKH:
        return f'Addresses start with {_bech32_start(coins[coin_symbol])}'
    start_address, end_address = _base58_range(coin_symbol, address_type)
    return f'Addresses are from {start_address[:4]}… to {end_address[:4]}…'
```

```python
def _address_function(coin, address_type: str, compressed: bool):
    """Function calculating the address of the type from the public key (point)."""
    if address_type == P2WPKH:
        hrp = coin['bech32_hrp']
        return lambda point: encode(hrp, 0, calculate_public_key_hash(point))
    segwit = address_type == P2SH_P2WPKH
    prefix = coin['script_prefix_bytes' if segwit else 'address_prefix_bytes']
    return lambda point: calculate_address(
        calculate_public_key_hash(point, compressed, segwit), prefix
    )


def find_address(
    coin_symbol: str, address_type: str, pattern: str, compressed: bool, stop, attempts=None
) -> Optional[Tuple[str, str]]:
    """Check consecutive keys from a random one until the address starts with the pattern
    (return the address and WIF) or stop is set (return None). The attempts counter
    (multiprocessing.Value) is increased every ATTEMPTS_PER_UPDATE keys."""
    coin = coins[coin_symbol]
    address_of = _address_function(coin, address_type, compressed)
    seed = secp256k1.gen_private_key()
    point = seed * secp256k1.G
    counter = 0
    while True:
        address = address_of(point)
        if address.startswith(pattern):
            private_key = (seed + counter) % secp256k1.n
            wif = convert_private_key_to_wif_format(
                private_key, coin['secret_prefix_bytes'], compressed
            )
            return address, wif
        point += secp256k1.G
        counter += 1
        if counter % ATTEMPTS_PER_UPDATE == 0:
            if attempts is not None:
                with attempts.get_lock():
                    attempts.value += ATTEMPTS_PER_UPDATE
            if stop.is_set():
                return None


def search_worker(coin_symbol, address_type, pattern, compressed, stop, attempts, results):
    """Process searching for the address, the result is put into the results queue."""
    found = find_address(coin_symbol, address_type, pattern, compressed, stop, attempts)
    if found:
        results.put(found)
        stop.set()


class VanitySearch:
    """Search for the vanity address in processes. They are started with spawn, which is safe in
    programs with threads (e.g. Qt GUI), and the result is passed back in memory."""

    def __init__(
        self,
        coin_symbol: str,
        address_type: str,
        pattern: str,
        workers: int,
        compressed: bool = True,
    ):
        if not compressed and address_type != P2PKH:
            raise ValueError('Segwit addresses require compressed public key')
        self.pattern = validate_vanity_pattern(pattern, coin_symbol, address_type)
        self.coin_symbol = coin_symbol
        self.address_type = address_type
        self.workers = workers
        self.compressed = compressed
        self._context = multiprocessing.get_context('spawn')
        self._stop = self._context.Event()
        self._attempts = self._context.Value('Q', 0)
        self._results = self._context.Queue()
        self._processes = []
        self._result = None

    def start(self) -> None:
        for _ in range(self.workers):
            process = self._context.Process(
                target=search_worker,
                args=(
                    self.coin_symbol,
                    self.address_type,
                    self.pattern,
                    self.compressed,
                    self._stop,
                    self._attempts,
                    self._results,
                ),
                daemon=True,
            )
            process.start()
            self._processes.append(process)

    @property
    def attempts(self) -> int:
        return self._attempts.value

    @property
    def running(self) -> bool:
        return any(process.is_alive() for process in self._processes)

    @property
    def failed(self) -> bool:
        """Some process ended with an error before the address was found or the search stopped."""
        return not self._stop.is_set() and any(
            process.exitcode not in (None, 0) for process in self._processes
        )

    def result(self) -> Optional[Tuple[str, str]]:
        """Found address and private key in WIF format or None, it does not block."""
        if self._result is None:
            try:
                self._result = self._results.get_nowait()
            except queue.Empty:
                pass
        return self._result

    def stop(self) -> None:
        self._stop.set()
        self.result()
        for process in self._processes:
            process.join(timeout=5)
            if process.is_alive():
                process.terminate()
                process.join()
        self.result()
```

def print_script_address(coin_symbol, input_script, output_dir):
    try:
        validate_hex_script(input_script)
    except ValueError as exc:
        logger.error(exc)
        return
    address = convert_script_to_p2sh_address(

    )
    if output_dir:

def start_workers(args):
    coin_symbol = args.symbol
    if args.input_script:
        print_script_address(coin_symbol, args.input_script, args.output_dir)
        return
    address_type = P2SH_P2WPKH if args.segwit else args.address_type
    try:
        search = VanitySearch(
            coin_symbol, address_type, args.pattern, args.cores, not args.uncompressed
        )
    except ValueError as exc:
        logger.error(exc)
        return
    print(
        f'Looking for pattern {search.pattern} for {coins[coin_symbol]["name"]} '
        f'using {args.cores} workers'
    )
    search.start()
    started = last_report = time.monotonic()
    try:
        while search.result() is None:
            if search.failed:
                logger.error('Vanity search process ended unexpectedly')
                return
            time.sleep(0.1)
            if time.monotonic() - last_report >= PROGRESS_SECONDS:
                last_report = time.monotonic()
                speed = search.attempts / (last_report - started)
                print(f'checked {search.attempts / 1000000:.1f}M addresses ({speed:.0f}/sec)')
    finally:
        search.stop()
    address, wif_private_key = search.result()
    print(f'Address: {address}\nPrivate key: {wif_private_key}')
    if args.output_dir:
        save_qrcode(address, args.output_dir, error_correct='L')
        save_qrcode(wif_private_key, args.output_dir, f'{address}_private_key.png')
        print(f'QR codes were saved in directory {args.output_dir}')


def main():

    start_workers(args)

def translate_address_text(text: str, input_symbol: str, output_symbol: str) -> str:
    """Address of the same public key hash (or script hash) for another coin."""
    address = parse_address(text, input_symbol)
    if detect_address_kind(address, input_symbol) == 'p2sh':
        key = 'script_prefix_bytes'
    else:
        key = 'address_prefix_bytes'
    return translate_address(address, coins[input_symbol][key], coins[output_symbol][key])


def parse_private_key_input(text: str, integer: bool) -> str:
    """Private key in WIF format of any coin, or an integer (decimal or hex starting with 0x)
    converted to WIF of bitcoin with compressed public key."""
    value = text.strip()
    if not value:
        raise ValueError('Enter the private key')
    if integer:
        number = get_integer(value.lower())
        if number is None or not 0 < number < secp256k1.n:
            raise ValueError(
                f'Private key must be an integer from 1 to {secp256k1.n - 1} (decimal or hex)'
            )
        return convert_private_key_to_wif_format(number, coins['BTC']['secret_prefix_bytes'])
    if set(value) - set(BASE58_ALPHABET):
        raise ValueError('Private key in WIF format can contain only base58 characters')
    data = base58_to_bytes(value)
    payload, checksum = data[:-4], data[-4:]
    if len(data) < 5 or double_sha256(payload).digest()[:4] != checksum:
        raise ValueError('Incorrect checksum of the private key')
    compressed = len(payload) == 34 and payload[-1] == 1
    if len(payload) != 33 and not compressed:
        raise ValueError('Incorrect private key')
    private_key, _ = get_private_key_from_wif_format(value)
    if not 0 < private_key < secp256k1.n:
        raise ValueError('Incorrect private key')
    return value


def translated_key_addresses(wif: str, output_symbol: str) -> Tuple[str, List[Tuple[str, str]]]:
    """Private key in WIF format of another coin and its addresses of all supported types."""
    coin = coins[output_symbol]
    translated = translate_private_key(wif, coin['secret_prefix_bytes'])
    private_key, compressed = get_private_key_from_wif_format(wif)
    address_types = get_address_types(coin) if compressed else (P2PKH,)
    return translated, [
        (
            address_type,
            convert_private_key_to_coin_address(private_key, coin, compressed, address_type),
        )
        for address_type in address_types
    ]


def parse_vanity_pattern(text: str, coin_symbol: str, address_type: str) -> str:
    pattern = text.strip()
    if not pattern:
        raise ValueError('Enter the beginning of the address')
    return validate_vanity_pattern(pattern, coin_symbol, address_type)


def format_duration(seconds: float) -> str:
    for unit, size in (('days', 86400), ('h', 3600), ('min', 60)):
        if seconds >= size:
            return f'{seconds / size:.0f} {unit}'
    return f'{seconds:.0f} s'

class VanityTab(QWidget):
    def __init__(self, app):
        super().__init__()
        self.app = app
        self.search: Optional[VanitySearch] = None
        self.started = 0.0
        self.speed = DEFAULT_SPEED

        cores = os.cpu_count() or 1
        self.coin = QComboBox()

        self.address_type = QComboBox()

        self.pattern_status = QLabel()

        pass  # the tab does not show wallet addresses

        self.discard()

    def _coin_changed(self):
        address_types = [
            address_type
            for address_type in get_address_types(coins[self.coin.currentText()])
            if address_type in VANITY_ADDRESS_TYPES
        ]
        self.address_type.blockSignals(True)
        self.address_type.clear()
        self.address_type.addItems(address_types)
        self.address_type.blockSignals(False)
        self._validate_pattern()

    def _validate_pattern(self) -> Optional[str]:
        """Show the hint and the estimated time, return the pattern or None when incorrect."""
        coin_symbol, address_type = self.coin.currentText(), self.address_type.currentText()
        hint = pattern_hint(coin_symbol, address_type)
        try:
            pattern = parse_vanity_pattern(self.pattern.text(), coin_symbol, address_type)
        except ValueError as error:
            self.pattern_status.setText(
                f'{html.escape(hint)}<br><span style="color: {INVALID_COLOR}">'
                f'{html.escape(str(error))}</span>'
            )
            self.start_button.setEnabled(False)
            return None
        attempts = expected_attempts(pattern, coin_symbol, address_type)
        seconds = attempts / (self.speed * self.workers.value())
        estimate = f'~{attempts:,} attempts, ~{format_duration(seconds)}'
        if attempts > LONG_SEARCH_ATTEMPTS:
            estimate += ' (this may take very long)'
        self.pattern_status.setText(f'{html.escape(hint)}<br>{html.escape(estimate)}')
        self.start_button.setEnabled(self.search is None)
        return pattern

    def start(self):
        pattern = self._validate_pattern()
        if pattern is None or self.search is not None:
            return
        try:
            search = VanitySearch(
                self.coin.currentText(),
                self.address_type.currentText(),
                pattern,
                self.workers.value(),
            )
            search.start()
        except EXPECTED_ERRORS as error:
            self.app.show_error(error)
            return
        self.discard()
        self.search = search
        self.started = time.monotonic()
        self.start_button.setEnabled(False)
        self.stop_button.setEnabled(True)
        self.progress.setText('Starting processes...')
        self.timer.start(1000)

    def stop(self):
        if self.search is not None:
            self._stop_search()
            self.progress.setText(self.progress.text() + ' (stopped)')

    def _stop_search(self):
        self.timer.stop()
        if self.search is not None:
            self.search.stop()
            self.search = None
        self.stop_button.setEnabled(False)
        self._validate_pattern()

    def _poll(self):
        search = self.search
        if search is None:
            return
        elapsed = time.monotonic() - self.started
        attempts = search.attempts
        if attempts:
            self.speed = attempts / elapsed / search.workers
            self.progress.setText(
                f'Checked {attempts:,} addresses in {format_duration(elapsed)} '
                f'({attempts / elapsed:,.0f}/s)'
            )
        result = search.result()
        if result is not None:
            self._stop_search()
            self._show_result(search, *result)
        elif search.failed:
            self._stop_search()
            self.app.show_error(ValueError('Vanity search process ended unexpectedly'))

    def _show_result(self, search: VanitySearch, address: str, wif: str):
        self.found = (search.coin_symbol, search.address_type, address, wif)
        self.result_address.setText(address)
        self.result_key.setText(wif)
        self.result_label.setText(f'Vanity {search.pattern}')
        self.result_group.show()

    def add_to_wallet(self):
        if self.found is None or self.app.wallet is None:
            return
        coin_symbol, address_type, _, wif = self.found
        label = self.result_label.text().strip()
        try:
            if not label:
                raise ValueError('Enter the label')
            self.app.modify_wallet(
                lambda wallet: wallet.add_wif(label, coin_symbol, wif, address_type)
            )
        except EXPECTED_ERRORS as error:
            self.app.show_error(error)
            return
        self.discard()

    def discard(self):
        self.found = None
        self.result_address.clear()
        self.result_key.clear()
        self.result_label.clear()
        self.result_group.hide()
