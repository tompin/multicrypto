```python
    """Calculate address of the given type for the public key. P2TR is not supported, its output
    key is calculated from the private key (taproot_output_key)."""
    validate_address_type(address_type, coin)
    if address_type == P2TR:

    return encode(coin['bech32_hrp'], 0, calculate_public_key_hash(public_key))
```

```python
def add_magic_prefix(message: str, coin) -> bytes:

    name = coin.get('message_prefix', coin['name'].capitalize())
```

```python
def _recover_public_key(message, recid, r, s, curve=secp256k1, hash_function=double_sha256):
    """Public key which signed the message, None when the signature has no key with the recovery
    id. Calculation is based on http://www.secg.org/sec1-v2.pdf chapter 4.1.6
    'Public Key Recovery Operation'."""
    x = r + (recid // 2) * curve.n
    if x >= curve.p:
        return None
    alpha = (x**3 + curve.a * x + curve.b) % curve.p
    beta = modular_sqrt(alpha, curve.p)
    if beta * beta % curve.p != alpha:
        return None
    y = beta if (beta - recid) % 2 == 0 else curve.p - beta
    R = Point(curve, x, y)
    e = int.from_bytes(hash_function(message).digest(), byteorder='big')
    return modular_inverse(r, curve.n) * (s * R - e * curve.G)


def sign_message(
    coin,
    message: str,
    wif_private_key: str,
    address_type: str = P2PKH,
    curve=secp256k1,
    hash_function=double_sha256,
) -> bytes:
    """Sign the message with the private key of the address (BIP-137), return base64 signature.
    The first byte of the signature tells the address type and the recovery id of the key."""
    if address_type == P2TR:
        raise ValueError('Taproot addresses cannot sign messages (BIP-137)')
    private_key, compressed = get_private_key_from_wif_format(wif_private_key)
    public_key = curve.G * private_key
    # raises ValueError when the coin or the key does not support the address type
    convert_public_key_to_coin_address(public_key, coin, compressed, address_type)
    header = SIGNATURE_HEADERS[address_type] if compressed else UNCOMPRESSED_P2PKH_HEADER
    message_with_magic_prefix = add_magic_prefix(message, coin)
    r, s = sign(message_with_magic_prefix, private_key, curve, hash_function)
    signature = r.to_bytes(curve.bytes_size, byteorder='big') + s.to_bytes(
        curve.bytes_size, byteorder='big'
    )
    for recid in range(4):
        recovered = _recover_public_key(
            message_with_magic_prefix, recid, r, s, curve, hash_function
        )
        if recovered is not None and recovered == public_key:
            return b64encode(bytes([header + recid]) + signature)
    raise ValueError('Cannot sign the message')


def verify_signature(
    coin, message: str, signature: str, address: str, curve=secp256k1, hash_function=double_sha256
) -> bool:
    """Check the base64 signature (BIP-137) of the message made with the key of the address.
    The address can have any type of the recovered key, since some wallets (Electrum, Trezor)
    sign messages of segwit addresses with the header of P2PKH address."""
    try:
        data = b64decode(signature.strip(), validate=True)
    except binascii.Error:
        raise ValueError('Signature is not correct base64') from None
    size = 1 + 2 * curve.bytes_size
    if len(data) != size:
        raise ValueError(f'Signature must have {size} bytes, it has {len(data)}')
    header = data[0]
    if not UNCOMPRESSED_P2PKH_HEADER <= header < SIGNATURE_HEADERS[P2WPKH] + 4:
        raise ValueError(f'Incorrect first byte of the signature: {header}')
    compressed = header >= SIGNATURE_HEADERS[P2PKH]
    recid = (header - UNCOMPRESSED_P2PKH_HEADER) % 4
    r = int.from_bytes(data[1 : 1 + curve.bytes_size], byteorder='big')
    s = int.from_bytes(data[1 + curve.bytes_size :], byteorder='big')
    if not (0 < r < curve.n and 0 < s < curve.n):
        return False
    public_key = _recover_public_key(
        add_magic_prefix(message, coin), recid, r, s, curve, hash_function
    )
    if public_key is None:
        return False
    if compressed:
        address_types = [type_ for type_ in get_address_types(coin) if type_ != P2TR]
    else:
        address_types = [P2PKH]
    return any(
        convert_public_key_to_coin_address(public_key, coin, compressed, address_type) == address
        for address_type in address_types
    )
```

```python
def prepare_sweep(
    coin: Coin,
    utxos: List[Utxo],
    destination_address: str,
    fee: int,
    batch_size: int,
) -> List[PreparedTransaction]:
    """Prepare transactions sending all inputs to the address without change, each transaction
    spends at most batch_size inputs and pays the fee. Nothing is prepared when any batch cannot
    be sent."""
    utxos = [utxo for utxo in utxos if utxo]
    if not utxos:
        raise ValueError('No inputs found for the private key')
    dust_threshold = get_dust_threshold(address_to_script(destination_address, coin))
    prepared = []
    for start in range(0, len(utxos), batch_size):
        batch = utxos[start : start + batch_size]
        satoshis = sum(utxo['satoshis'] for utxo in batch)
        if satoshis <= fee:
            raise ValueError(f'Fee {fee} is not smaller than sum of batch inputs {satoshis}')
        if satoshis - fee < dust_threshold:
            raise ValueError(
                f'Amount {satoshis - fee} of the batch is below the dust threshold {dust_threshold}'
            )
        prepared.append(prepare_utxos(coin, batch, destination_address, satoshis - fee, fee))
    return prepared
```

```python
def describe_sweep(
    prepared: List[PreparedTransaction],
    coin_symbol: str,
    source_address: str,
    destination_address: str,
) -> List[Tuple[str, str]]:
    """Rows of the summary shown before the sweep transactions are broadcast."""

    def amount(satoshis: int) -> str:
        return f'{format_amount(satoshis)} {coin_symbol}'

    batches = '\n'.join(
        f'{number}. inputs: {len(transaction.inputs)}, '
        f'amount: {amount(transaction.outputs[0]["satoshis"])}, '
        f'fee: {amount(transaction.fee)}, size: {len(transaction.raw_transaction) // 2} bytes'
        for number, transaction in enumerate(prepared, start=1)
    )
    inputs = sum(len(transaction.inputs) for transaction in prepared)
    input_satoshis = sum(transaction.input_satoshis for transaction in prepared)
    return [
        ('From', source_address),
        ('To', destination_address),
        ('Transactions', str(len(prepared))),
        ('Inputs', f'{inputs} ({amount(input_satoshis)})'),
        ('Fee', amount(sum(transaction.fee for transaction in prepared))),
        ('Received', amount(sum(transaction.outputs[0]['satoshis'] for transaction in prepared))),
        ('Batches', batches),
```

```python
import pyqrcode
from PySide6.QtCore import Qt
from PySide6.QtGui import QImage, QPainter, QPixmap

    QApplication,
    QDialog,
    QDialogButtonBox,

    QVBoxLayout,

from multicrypto.wallet import WalletAddress

QUIET_ZONE = 4  # white modules around the code required by the QR code specification


def qr_image(text: str, scale: int = 6) -> QImage:
    matrix = pyqrcode.create(text, error='M').code
    size = (len(matrix) + 2 * QUIET_ZONE) * scale
    image = QImage(size, size, QImage.Format.Format_RGB32)
    image.fill(Qt.GlobalColor.white)
    painter = QPainter(image)
    for row, modules in enumerate(matrix):
        for column, module in enumerate(modules):
            if module:
                painter.fillRect(
                    (column + QUIET_ZONE) * scale,
                    (row + QUIET_ZONE) * scale,
                    scale,
                    scale,
                    Qt.GlobalColor.black,
                )
    painter.end()
    return image


class QrDialog(QDialog):

        super().__init__(parent)
        self.setWindowTitle(f'QR code of "{address.label}"')
        image = QLabel()
        image.setPixmap(QPixmap.fromImage(qr_image(address.address)))
        image.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.address_label = monospace_label(address.address)
        self.address_label.setTextInteractionFlags(Qt.TextInteractionFlag.TextSelectableByMouse)
        self.address_label.setAlignment(Qt.AlignmentFlag.AlignCenter)
        copy_button = QPushButton('Copy')
        copy_button.clicked.connect(lambda: QApplication.clipboard().setText(address.address))
        buttons = QDialogButtonBox(QDialogButtonBox.StandardButton.Close)
        buttons.addButton(copy_button, QDialogButtonBox.ButtonRole.ActionRole)

        layout = QVBoxLayout(self)
        layout.addWidget(
            QLabel(f'{address.coin} {address.address_type} · {address.path or "imported"}')
        )
        layout.addWidget(image)
        layout.addWidget(self.address_label)
        layout.addWidget(buttons)
```

```python
            lambda prepared: self._confirm(
                coin, coin_symbol, prepared, source_address, destination
            ),
            self._failed,
        )

    def _confirm(self, coin, coin_symbol, prepared, source_address, destination):
        self.prepare_button.setEnabled(True)
        if self.app.wallet is None:
            return
        rows = describe_sweep(prepared, coin_symbol, source_address, destination)
        if confirm(self, 'Confirm sweep', rows, 'Send'):
            self.prepare_button.setEnabled(False)
            self._broadcast(coin, prepared, [])

    def _broadcast(self, coin, prepared, txids: List[str]):
        """Broadcast the next transaction, the transactions are sent one after another."""
        if self.app.wallet is None:
            self.prepare_button.setEnabled(True)
            self.progress.setText('')
            return
        if len(txids) == len(prepared):
            self._sent(txids)
            return
        self.progress.setText(f'Broadcast {len(txids) + 1}/{len(prepared)}')
        transaction = prepared[len(txids)]
        self.app.worker.run(
            lambda: extract_txid(broadcast_prepared(coin, transaction)),
            lambda txid: self._broadcast(coin, prepared, txids + [txid]),
            lambda error: self._broadcast_failed(error, txids),
        )

    def _sent(self, txids: List[str]):
        self.prepare_button.setEnabled(True)
        self.progress.setText('')
        self.wif.clear()
        show_lines(self, 'Sweep sent', f'Transactions sent: {len(txids)}', txids)

    def _broadcast_failed(self, error, txids: List[str]):
        self.prepare_button.setEnabled(True)
        self.progress.setText('')
        if self.app.wallet is None:
            return
        message = describe_error(error)
        if txids:
            message += '\n\nTransactions sent before the error:\n' + '\n'.join(txids)
        QMessageBox.critical(self, 'Error', message)

    def _failed(self, error):
        self.prepare_button.setEnabled(True)
        self.app.show_error(error)
```
