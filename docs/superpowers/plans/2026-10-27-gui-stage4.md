def prepare_with_fee_rate(
    coin: Coin,
    utxos: Iterable[Utxo],
    destination_address: str,
    satoshis: int,
    fee_rate: float,
    change_address: Optional[str] = None,
) -> PreparedTransaction:
    """Prepare the transaction paying at least fee_rate satoshis per virtual byte. The fee depends
    on the size, which depends on the inputs selected for the fee, so the transaction is prepared
    again until the fee matches its size."""
    utxos = list(utxos)
    best = None
    fee = fee_for_rate(fee_rate, INITIAL_VSIZE)
    for _ in range(FEE_RATE_ATTEMPTS):
        prepared = prepare_utxos(coin, utxos, destination_address, satoshis, fee, change_address)
        required = fee_for_rate(fee_rate, prepared.vsize)
        if prepared.fee >= required:
            best = prepared
            if prepared.fee == required:
                break
        fee = required
    if best is None:
        raise ValueError(f'Cannot calculate the fee for the rate {fee_rate} sat/vB')
    return best
```

def _sweep_fee(coin: Coin, batch: List[Utxo], destination_address: str, fee_rate: float) -> int:
    """Fee paying the rate for the transaction sending all inputs of the batch."""
    satoshis = sum(utxo['satoshis'] for utxo in batch)
    fee = 0
    for _ in range(FEE_RATE_ATTEMPTS):
        if fee >= satoshis:
            break
        vsize = prepare_utxos(coin, batch, destination_address, satoshis - fee, fee).vsize
        required = fee_for_rate(fee_rate, vsize)
        if fee >= required:
            break
        fee = required
    return fee
```

    response = http_request(
        f'{self.base_url}/addr/{address}', params={'noTxList': 1}, timeout=self.timeout
    )
    if not response.ok:
        raise ValueError(f'API error: {response.text}')
    data = response.json()
    return data['txApperances'] + data.get('unconfirmedTxApperances', 0)


def estimate_fee_rates(self) -> Dict[int, float]:
    """Fee rates in satoshis per virtual byte for the confirmation targets."""
    response = http_request(
        f'{self.base_url}/utils/estimatefee',
        params={'nbBlocks': ','.join(map(str, FEE_TARGETS))},
        timeout=self.timeout,
    )
    if not response.ok:
        raise ValueError(f'API error: {response.text}')
    # coins per kilobyte, -1 when the estimate is not available
    return {

        for blocks, rate in response.json().items()
        if rate > 0
    }

        id=f'{account.id}/{hd_address.chain}/{hd_address.index}',
        label=hd_address.label,
        coin=account.coin,
        address_type=account.address_type,
        address=self.address_at(account, hd_address.chain, hd_address.index),
        path=f'{account.path}/{hd_address.chain}/{hd_address.index}',
        account_id=account.id,
        index=hd_address.index,
        chain=hd_address.chain,
    )


@staticmethod
def _next_index(account: HDAccount, chain: int) -> int:
    indexes = [address.index for address in account.addresses if address.chain == chain]
    index = max(indexes, default=-1) + 1
    if index >= HARDENED:
        raise ValueError('There are no more addresses in the account')
    return index


def new_address(self, account_id: str, label: str = '') -> WalletAddress:
    """Add the next receiving address of the account."""
    account = self.get_account(account_id)
    index = self._next_index(account, RECEIVE_CHAIN)
    return self.add_address(account_id, RECEIVE_CHAIN, index, label)


def next_change_address(self, account_id: str) -> WalletAddress:
    """Next change address of the account, it is not added to the wallet."""
    account = self.get_account(account_id)
    index = self._next_index(account, CHANGE_CHAIN)
    return self._hd_address(
        account, HDAddress(index, f'{CHAIN_LABELS[CHANGE_CHAIN]} {index}', chain=CHANGE_CHAIN)
    )


def add_address(self, account_id: str, chain: int, index: int, label: str = '') -> WalletAddress:
    """Add the address of the account chain, the present address is returned unchanged."""
    account = self.get_account(account_id)
    for hd_address in account.addresses:
        if (hd_address.chain, hd_address.index) == (chain, index):
            return self._hd_address(account, hd_address)
    hd_address = HDAddress(index, label or f'{CHAIN_LABELS[chain]} {index}', chain=chain)
    wallet_address = self._hd_address(account, hd_address)  # fails when the wallet is locked
    account.addresses.append(hd_address)
    return wallet_address


def account_addresses(self, account: HDAccount) -> List[WalletAddress]:
    addresses = sorted(account.addresses, key=lambda address: (address.chain, address.index))
    return [self._hd_address(account, hd_address) for hd_address in addresses]

"""Discovery of used accounts and addresses of the HD wallet (BIP-44 account discovery)."""

from concurrent.futures import ThreadPoolExecutor
from dataclasses import dataclass, field
from typing import Callable, List, Optional

from multicrypto.api import get_api
from multicrypto.coins import coins, get_address_types
from multicrypto.wallet import CHANGE_CHAIN, RECEIVE_CHAIN, HDAccount, Wallet

GAP_LIMIT = 20  # unused addresses after the last used one ending the scan of the chain
THREADS = 4
MAX_ACCOUNTS = 100
MAX_ADDRESSES = 10000


class Cancelled(Exception):
    """Raised when the discovery has been cancelled."""


@dataclass
class DiscoveredAccount:
    coin: str
    address_type: str
    account: int
    receive: List[int] = field(default_factory=list)
    change: List[int] = field(default_factory=list)


def scan_chain(
    address_at: Callable[[int], str],
    tx_count: Callable[[str], int],
    progress: Callable[[int], None],
    cancelled: Callable[[], bool],
    mapper: Callable = map,
) -> List[int]:
    """Indexes of the used addresses of the chain. Windows of addresses are checked (with the
    mapper, e.g. map of the thread pool) until GAP_LIMIT addresses after the last used one are
    unused."""
    used: List[int] = []
    start = 0
    end = GAP_LIMIT
    while start < end:
        if cancelled():
            raise Cancelled()
        if end > MAX_ADDRESSES:
            raise ValueError(f'Stopped after {MAX_ADDRESSES} addresses of the chain')
        indexes = range(start, end)
        addresses = [address_at(index) for index in indexes]
        counts = list(mapper(tx_count, addresses))
        progress(len(indexes))
        used.extend(index for index, count in zip(indexes, counts) if count)
        start, end = end, (used[-1] + 1 if used else 0) + GAP_LIMIT
    return used


def discover(
    wallet: Wallet,
    coin_symbol: str,
    progress: Callable[[int], None],
    cancelled: Callable[[], bool],
    tx_count: Optional[Callable[[str], int]] = None,
) -> List[DiscoveredAccount]:
    """Used accounts of all address types of the coin: accounts are checked from 0 to the first
    one without transactions on the receiving and change chains."""
    coin = coins[coin_symbol]
    if 'slip44' not in coin or not coin.get('apis'):
        raise ValueError(f'{coin_symbol} cannot be scanned, it has no SLIP-44 coin type or API')
    if tx_count is None:
        tx_count = get_api(coin).get_tx_count
    found = []
    with ThreadPoolExecutor(THREADS) as executor:
        for address_type in get_address_types(coin):
            for number in range(MAX_ACCOUNTS + 1):
                if number == MAX_ACCOUNTS:
                    raise ValueError(
                        f'Stopped after {MAX_ACCOUNTS} accounts of {coin_symbol} {address_type}'
                    )
                account = HDAccount('', coin_symbol, address_type, number)
                chains = [
                    scan_chain(
                        lambda index, chain=chain: wallet.address_at(account, chain, index),
                        tx_count,
                        progress,
                        cancelled,
                        executor.map,
                    )
                    for chain in (RECEIVE_CHAIN, CHANGE_CHAIN)
                ]
                if not any(chains):
                    break
                found.append(DiscoveredAccount(coin_symbol, address_type, number, *chains))
    return found


def add_discovered(wallet: Wallet, discovered: List[DiscoveredAccount]) -> int:
    """Add the discovered accounts and addresses missing in the wallet, return the number of the
    added addresses."""
    added = 0
    for item in discovered:
        account = next(
            (
                account
                for account in wallet.accounts
                if (account.coin, account.address_type, account.account)
                == (item.coin, item.address_type, item.account)
            ),
            None,
        )
        if account is None:
            account = wallet.add_account(
                f'{item.coin} {item.address_type} #{item.account}',
                item.coin,
                item.address_type,
                item.account,
            )
        present = {(address.chain, address.index) for address in account.addresses}
        for chain, indexes in ((RECEIVE_CHAIN, item.receive), (CHANGE_CHAIN, item.change)):
            for index in indexes:
                if (chain, index) not in present:
                    wallet.add_address(account.id, chain, index)
                    added += 1
    return added

class FeeRateField(QWidget):
    """Fee rate in sat/vB, Estimate fills it with the rate for the selected number of blocks."""

    def __init__(self, app, coin_symbol: Callable[[], Optional[str]]):
        super().__init__()
        self.app = app
        self.coin_symbol = coin_symbol
        self.rate = QLineEdit()
        self.rate.setPlaceholderText('sat/vB')
        self.target = QComboBox()

        self.target.setCurrentIndex(self.target.findData(DEFAULT_TARGET))
        self.button = QPushButton('Estimate')
        self.button.clicked.connect(lambda _checked=False: self.estimate())
        self.status = QLabel()
        layout = QHBoxLayout(self)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.addWidget(self.rate)
        layout.addWidget(QLabel('confirmation in'))
        layout.addWidget(self.target)
        layout.addWidget(self.button)
        layout.addWidget(self.status, 1)
        self.rate.textEdited.connect(lambda _text: self.status.setText(''))

            self._error('Select the source address first')
            return

        self.status.setText('')
        self.button.setEnabled(False)
        self.app.worker.run(
            lambda: estimate_fee_rates(coins[coin_symbol]),
            lambda rates: self._estimated(rates, target),
            self._failed,
        )

    def _estimated(self, rates: Dict[int, float], target: int):
        self.button.setEnabled(True)
        targets = sorted(rates)
        later = [blocks for blocks in targets if blocks >= target]
        rate = rates[later[0]] if later else rates[targets[-1]]
        self.rate.setText(f'{math.ceil(rate * 10) / 10:.1f}')

    def _failed(self, error):
        self.button.setEnabled(True)
        self._error(describe_error(error))

    def _error(self, text: str):
        self.status.setText(f'<span style="color: {INVALID_COLOR}">{html.escape(text)}</span>')

```python
"""Dialog scanning the blockchain for used accounts and addresses of the HD wallet."""

import threading
from typing import List

from PySide6.QtCore import QTimer
from PySide6.QtWidgets import (
    QCheckBox,
    QDialog,
    QGridLayout,
    QHBoxLayout,
    QLabel,
    QPlainTextEdit,
    QPushButton,
    QVBoxLayout,
)

from multicrypto.discovery import Cancelled, DiscoveredAccount, add_discovered, discover
from multicrypto.gui.forms import EXPECTED_ERRORS, coins_with_api, coins_with_slip44, describe_error

COLUMNS = 6


def scannable_coins() -> List[str]:
    return sorted(set(coins_with_slip44()) & set(coins_with_api()))


class ScanDialog(QDialog):
    def __init__(self, app):
        super().__init__(app)
        self.app = app
        self.setWindowTitle('Scan for used addresses')
        self.discovered: List[DiscoveredAccount] = []
        self.checked = 0
        self.lock = threading.Lock()
        self.cancel_requested = False
        self.coins = {symbol: QCheckBox(symbol) for symbol in scannable_coins()}
        self.coins['BTC'].setChecked(True)
        grid = QGridLayout()
        for position, box in enumerate(self.coins.values()):
            grid.addWidget(box, position // COLUMNS, position % COLUMNS)
        self.status = QLabel(
            'Addresses of all types and accounts (BIP-44) are checked until 20 unused ones.'
        )
        self.results = QPlainTextEdit()
        self.results.setReadOnly(True)
        self.scan_button = QPushButton('Scan')
        self.scan_button.clicked.connect(lambda _checked=False: self.scan())
        self.add_button = QPushButton('Add to wallet')
        self.add_button.setEnabled(False)
        self.add_button.clicked.connect(lambda _checked=False: self.add())
        close_button = QPushButton('Close')
        close_button.clicked.connect(self.reject)
        buttons = QHBoxLayout()
        buttons.addWidget(self.scan_button)
        buttons.addWidget(self.add_button)
        buttons.addStretch()
        buttons.addWidget(close_button)
        layout = QVBoxLayout(self)
        layout.addWidget(QLabel('Coins'))
        layout.addLayout(grid)
        layout.addWidget(self.status)
        layout.addWidget(self.results)
        layout.addLayout(buttons)
        self.timer = QTimer(self)
        self.timer.timeout.connect(self._show_progress)
        self.resize(640, 420)

    def _progress(self, count: int):
        with self.lock:  # called from the threads of the worker
            self.checked += count

    def _show_progress(self):
        self.status.setText(f'Checked {self.checked} addresses...')

    def scan(self):
        symbols = [symbol for symbol, box in self.coins.items() if box.isChecked()]
        wallet = self.app.wallet
        if not symbols or wallet is None:
            self.status.setText('Select at least one coin')
            return
        self.checked = 0
        self.cancel_requested = False
        self.discovered = []
        self.results.clear()
        self.add_button.setEnabled(False)
        self.scan_button.setEnabled(False)

        def cancelled():
            return self.cancel_requested or self.app.wallet is not wallet

        def run():
            results = []
            for symbol in symbols:
                try:
                    results.append((symbol, discover(wallet, symbol, self._progress, cancelled)))
                except Cancelled:
                    return None
                except EXPECTED_ERRORS as error:
                    results.append((symbol, describe_error(error)))
            return results

        self.timer.start(300)
        self.app.worker.run(run, self._scanned, self._failed)

    def _scanned(self, results):
        self.timer.stop()
        self.scan_button.setEnabled(True)
        if results is None or self.app.wallet is None:
            self.status.setText('Cancelled')
            return
        lines = []
        for symbol, result in results:
            if isinstance(result, str):
                lines.append(f'{symbol}: {result}')
                continue
            self.discovered.extend(result)
            lines.extend(
                f'{item.coin} {item.address_type} account {item.account}: '
                f'{len(item.receive)} receiving, {len(item.change)} change'
                for item in result
            )
        if not self.discovered:
            lines.append('No used addresses found')
        self.results.setPlainText('\n'.join(lines))
        self.status.setText(f'Checked {self.checked} addresses')
        self.add_button.setEnabled(bool(self.discovered))

    def _failed(self, error):
        self.timer.stop()
        self.scan_button.setEnabled(True)
        self.status.setText(describe_error(error))

    def add(self):
        if self.app.wallet is None:
            return
        try:
            added = self.app.modify_wallet(lambda wallet: add_discovered(wallet, self.discovered))
        except EXPECTED_ERRORS as error:
            self.app.show_error(error)
            return
        self.add_button.setEnabled(False)
        self.status.setText(f'Added {added} addresses to the wallet')

    def reject(self):
        self.cancel_requested = True
        super().reject()
```
